# FM-FAM v0.2.8: the key combinations of the randoms and the inits (nothing waits for a hold), NUDGE ALL, and the kick's
# nudge with OCT- held
#   fam/build/host/fm-fam_host fam/tests/v028a.txt DIRECTORY
wait 3000
# the start pattern: steps on 0 (accent), 2, 3, 6; the kick on 0 and 4
expect trig0 2
expect ktrig0 1
expect ktrig4 1
# SEQ with OCT- held: K3 is NUDGE ALL (the steps and the kick's line together), K4 the steps alone
tap SEQ
press OCT-
wait 100
expect nudge 0
expect nudge_all 0
spin K3 1
wait 60
expect nudge_all 1
expect nudge 0
expect trig0 0
expect trig1 2
expect ktrig0 0
expect ktrig1 1
expect ktrig5 1
shot seq_nudge.png
spin K4 1
wait 60
expect nudge 1
expect trig2 2
expect ktrig1 1
spin K4 -1
spin K3 -1
wait 60
expect trig0 2
expect ktrig0 1
expect ktrig4 1
release OCT-
wait 100
# the kick's sequence pages with OCT- held: K3 NUDGE ALL, K4 the kick's line alone; without OCT-, K4 of its second page does nothing
tap ARP
spin SELECT -9
wait 80
expect mode 3
expect kpg 3
press OCT-
wait 100
spin K4 1
wait 60
expect nudge 1
expect ktrig1 1
expect ktrig0 0
expect trig0 2
shot kick_nudge.png
spin K3 1
wait 60
expect nudge_all 1
expect ktrig2 1
expect trig1 2
spin K3 -1
spin K4 -1
wait 60
release OCT-
wait 100
expect ktrig0 1
expect trig0 2
spin SELECT 1
wait 80
expect kpg 4
spin K4 3
wait 60
expect ktrig0 1
expect ktrig4 1
# OCT- + GLO: this page is initialised (the kick's line: empty), and the mode stays
press OCT-
wait 60
tap GLO
wait 60
release OCT-
wait 100
expect mode 3
expect ktrig0 0
expect ktrig4 0
expect trig0 2
# on a page without an init nothing changes
tap SAVE
press OCT-
wait 60
tap GLO
wait 60
release OCT-
wait 100
expect mode 8
# OCT+ + GLO: the pattern starts over, its steps and its sound
tap SEQ
press OCT+
wait 60
tap GLO
wait 60
release OCT+
wait 100
expect mode 0
expect trig0 0
expect trig2 0
# OCT+ + REC: all, at once (REC still held)
press OCT+
wait 60
press REC
wait 100
expect kick6 >99
release REC
release OCT+
wait 100
expect mode 0
# OCT- + HOME: a melody; OCT+ + HOME: percussion (no attack, a short decay)
press OCT-
wait 60
tap HOME
wait 60
release OCT-
wait 100
expect sc3 1
expect syn10 0
press OCT+
wait 60
tap HOME
wait 60
release OCT+
wait 100
expect syn17 0
expect syn18 >34
# GLO alone still opens the patterns
tap GLO
expect mode 5
